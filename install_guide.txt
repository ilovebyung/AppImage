# install_guide.txt
# ------------------------------------------------------------
# Bundle a Streamlit application into a portable AppImage
# using appimagetool.  The process creates a self‑contained
# directory (AppDir) and then compresses it into a single
# executable.
# ------------------------------------------------------------

# 1. Create the AppDir structure
mkdir -p App.AppDir/usr/{bin,lib/python3.12,share/applications}

# 2. Copy the Python standard library (including the 'encodings' module)
#    – this resolves the “module encodings is missing” error
cp -r /usr/lib/python3.12/* App.AppDir/usr/lib/python3.12/   # [1]

# 3. Copy the project files (source code, requirements, etc.) **without**
#    duplicating the top‑level folder.  The contents of the project
#    directory are placed directly under `usr/poke/`.
PROJECT_ROOT=~/Documents/app          # adjust to your actual path
mkdir -p App.AppDir/usr/app
cp -r "$PROJECT_ROOT"/.* "$PROJECT_ROOT"/* App.AppDir/usr/app/   # [1]

# 4. Install the virtual‑environment dependencies into the AppDir
python3 -m venv App.AppDir/usr/venv
source App.AppDir/usr/venv/bin/activate
pip install -r "$PROJECT_ROOT/requirements.txt"

# 5. Add the Streamlit entry point (e.g., a wrapper script)
cat > App.AppDir/usr/bin/app <<'EOF'
#!/usr/bin/env bash
exec "$PWD/../venv/bin/streamlit" run "$PWD/../app/Home.py" \
    --server.headless true \
    --server.port 8888  "$@"
EOF
chmod +x App.AppDir/usr/bin/app

# 6. Add desktop file and icon so the AppImage integrates
cat > App.AppDir/usr/share/applications/app.desktop <<'EOF'
[Desktop Entry]
Name=App
Exec=app
Icon=app
Type=Application
Terminal=false
Categories=Utility;
EOF

# Optional
cp AppImage/AppRun AppImage/app.desktop App.AppDir/
cp AppImage/app.png App.AppDir/

cp App.AppDir/usr/app/.streamlit/credentials.yaml ~/.local/share/pos
cp App.AppDir/usr/app/pos.database ~/.local/share/pos

# 7. Build the AppImage
./appimagetool-x86_64.AppImage App.AppDir
